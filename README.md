import random

# Список цитат
quotes = [
    "Будь собой, остальные роли уже заняты. – Оскар Уайльд",
    "Не ошибается тот, кто ничего не делает. – Теодор Рузвельт",
    "Единственный способ делать отличную работу — любить то, что ты делаешь. – Стив Джобс",
    "Всегда помните, что ваше собственное решение — это сила, которая управляет вашей жизнью. – Аун Сан Су Чжи",
    "Успех — это способность идти от поражения к поражению, не теряя энтузиазма. – Уинстон Черчилль",
]

def get_random_quote():
    # Получаем случайный индекс из списка цитат
    index = random.randint(0, len(quotes) - 1)
    return quotes[index]

if __name__ == "__main__":
    quote = get_random_quote()
    print(quote)
